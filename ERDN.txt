local function LoadConfig()
    if type(readfile) == "function" and type(isfile) == "function" then
        pcall(function()
            if isfile(ConfigFileName) then
                local savedData = HttpService:JSONEncode(readfile(ConfigFileName))

                    AutoExecuteActive = savedData.AutoExecute == true
                    AutoRejoinActive = savedData.AutoRejoin == true
                    RedactNameActive = savedData.RedactName == true
                    AntiCheatActive = savedData.AntiCheat == true
                end
            end
        end)
    end
end

local nameVisibilityState = {Character = nil, Connections = {}, Guis = {}, Watches = {}}
local function clearNameVisibility()
    local state = nameVisibilityState
    for _, connection in ipairs(state.Connections) do connection:Disconnect() end
    table.clear(state.Connections)
    for gui, connection in pairs(state.Watches) do connection:Disconnect() state.Watches[gui] = nil end
    for gui, wasEnabled in pairs(state.Guis) do if gui.Parent then gui.Enabled = wasEnabled end end
    table.clear(state.Guis)
    state.Character = nil
end

local function applyNameVisibility()
    local state = nameVisibilityState
    local character = LP.Character
    if state.Character ~= character then
        clearNameVisibility()
        state.Character = character
        if character then
            local function watchNameGui(gui)
                if not gui:IsA("BillboardGui") or gui.Name ~= "nameGui" or state.Guis[gui] ~= nil then return end
                state.Guis[gui] = gui.Enabled
                state.Watches[gui] = gui:GetPropertyChangedSignal("Enabled"):Connect(function()
                    if RedactNameActive and gui.Enabled then gui.Enabled = false end
                end)
                if RedactNameActive then gui.Enabled = false end
            end
            table.insert(state.Connections, character.DescendantAdded:Connect(watchNameGui))
            table.insert(state.Connections, character.DescendantRemoving:Connect(function(gui)
                if state.Guis[gui] ~= nil then
                    local connection = state.Watches[gui]
                    if connection then connection:Disconnect() end
                    state.Watches[gui] = nil
                    gui.Enabled = state.Guis[gui]
                    state.Guis[gui] = nil
                end
            end))
            for _, gui in ipairs(character:GetDescendants()) do watchNameGui(gui) end
        end
    end
    for gui, wasEnabled in pairs(state.Guis) do
        if gui.Parent then gui.Enabled = not RedactNameActive and wasEnabled end
    end
end

LP.CharacterAdded:Connect(function()
    applyNameVisibility()
end)

local function applyPetLoadout(selected, isCurrent)
    local rEvents = ReplicatedStorage:FindFirstChild("rEvents")
    local remote = rEvents and rEvents:FindFirstChild("equipPetEvent")
    if not remote then return false end

    local petsFolder = LP:FindFirstChild("petsFolder")
    if petsFolder then
        for _, folder in ipairs(petsFolder:GetChildren()) do
            if folder:IsA("Folder") then
                for _, pet in ipairs(folder:GetChildren()) do
                    if isCurrent and not isCurrent() then return false end
                    remote:FireServer("unequipPet", pet)
                end
            end
        end
    end
    task.wait(0.05)
    for _, pet in ipairs(selected) do
        if isCurrent and not isCurrent() then return false end
        remote:FireServer("equipPet", pet)
    end
    return not isCurrent or isCurrent()
end

local function sendTrainingReps(state, rate, maxBurst, canSend)
    local now = os.clock()
    local elapsed = math.clamp(now - (state.LastAt or now), 0, 0.1)
    state.LastAt = now
    state.Tokens = math.min(maxBurst, (state.Tokens or 0) + rate * elapsed)
    local sent = 0
    local muscleEvent = LP:FindFirstChild("muscleEvent")
    while state.Tokens >= 1 and canSend() and muscleEvent do
        muscleEvent:FireServer("rep")
        state.Tokens = state.Tokens - 1
        sent = sent + 1
    end
    return sent
end

local function createFarmLoop(getState, rate, burst)
    task.spawn(function()
        local cachedEvent = LP:FindFirstChild("muscleEvent")
        LP.ChildAdded:Connect(function(child)
            if child.Name == "muscleEvent" then cachedEvent = child end
        end)

        local INTERVAL = burst / rate

        while true do
            if getState() then
                if not cachedEvent or not cachedEvent.Parent then
                    cachedEvent = LP:FindFirstChild("muscleEvent")
                end
                if cachedEvent then
                    local start = os.clock()
                    for i = 1, burst do
                        if not getState() then break end
                        pcall(function() cachedEvent:FireServer("rep") end)
                    end
                    local remaining = INTERVAL - (os.clock() - start)
                    if remaining > 0 then task.wait(remaining) else task.wait(0.02) end
                else
                    task.wait(0.05)
                end
            else
                task.wait(0.15)
            end
        end
    end)
end

createFarmLoop(function() return FastFarm800 end, 800, 60)

local function Create(className, properties, parent)
    local instance = Instance.new(className)
    for property, value in pairs(properties) do
        instance[property] = value
    end
    if parent then
        instance.Parent = parent
    end
    return instance
end

local function MakeDraggable(dragBar, object)
    local dragging, dragStart, startPos
    
    dragBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = object.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    
    UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType ~= Enum.UserInputType.MouseMovement and input.UserInputType ~= Enum.UserInputType.Touch then return end
        local delta = input.Position - dragStart
        object.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end)
end

local function formatNumber(number)
    local isNegative = number < 0
    number = math.abs(number)
    if number >= 1e18 then return (isNegative and "-" or "") .. string.format("%.2fQi", number / 1e18)
    elseif number >= 1e15 then return (isNegative and "-" or "") .. string.format("%.2fQa", number / 1e15)
    elseif number >= 1e12 then return (isNegative and "-" or "") .. string.format("%.2fT", number / 1e12)
    elseif number >= 1e9 then return (isNegative and "-" or "") .. string.format("%.2fB", number / 1e9)
    elseif number >= 1e6 then return (isNegative and "-" or "") .. string.format("%.2fM", number / 1e6)
    elseif number >= 1e3 then return (isNegative and "-" or "") .. string.format("%.2fK", number / 1e3)
    else return (isNegative and "-" or "") .. string.format("%.0f", number) end
end

local function formatCurrentStat(val)
    return tostring(val or 0)
end

local function getOwnedConfiguredPets()
    local ownedPets = {}
    local petsFolder = lp.player:FindFirstChild("petsFolder")
    if petsFolder then
        for _, folder in ipairs(petsFolder:GetChildren()) do
            if folder:IsA("Folder") then
                for _, pet in ipairs(folder:GetChildren()) do
                    local config = PETS[pet.Name]
                    if config then
                        table.insert(ownedPets, {
                            instance = pet,
                            name = pet.Name,
                            repSpeed = config.repSpeed or 0,
                            strength = config.strength or 0,
                            durability = config.durability or 0,
                            rebirth = config.rebirth or 0,
                        })
                    end
                end
            end
        end
    end
    return ownedPets
end

local function equipBestFarmingPets(isCurrent)
    if isCurrent and not isCurrent() then return end
    local pets = getOwnedConfiguredPets()
    local selected = {}
    local equipped = {}
    local equippedCount = 0

    if RebirthChaos then
        for _, pet in ipairs(pets) do
            if pet.name == "Swift Samurai" and equippedCount < MAX_PET_SLOTS then
                table.insert(selected, pet.instance)
                equipped[pet.instance] = true
                equippedCount = equippedCount + 1
            end
        end
    elseif RebirthsTitans then
        for _, pet in ipairs(pets) do
            if pet.name == "Omega Overlord" and equippedCount < MAX_PET_SLOTS then
                table.insert(selected, pet.instance)
                equipped[pet.instance] = true
                equippedCount = equippedCount + 1
            end
        end
    end

    table.sort(pets, function(a, b) return a.repSpeed > b.repSpeed end)
    for _, pet in ipairs(pets) do
        if equippedCount >= MAX_PET_SLOTS then break end
        if not equipped[pet.instance] and pet.repSpeed > 0 then
            table.insert(selected, pet.instance)
            equipped[pet.instance] = true
            equippedCount = equippedCount + 1
        end
    end

    applyPetLoadout(selected, isCurrent)
    return equippedCount
end

RebirthTab.updateAdaptiveSpeed = function(cycle, finalSample)
    local adaptive = RebirthTab.adaptive
    local now = os.clock()
    if not finalSample and now - cycle.SampleAt < 0.25 then return end
    local strength = RebirthTab.strength.Value
    local gain = strength - cycle.StartStrength
    if gain > 0 and cycle.Sent > 0 and (finalSample or strength > cycle.SampleStrength) then
        adaptive.StrengthPerRep = gain / cycle.Sent
        cycle.SampleStrength = strength
    end
    if adaptive.StrengthPerRep and adaptive.StrengthPerRep > 0 then
        local reps = math.max(1, math.ceil(math.max(0, cycle.Target - cycle.StartStrength) / adaptive.StrengthPerRep))
        adaptive.SendRate = math.clamp(reps / adaptive.TargetSeconds, adaptive.MinRate, adaptive.MaxRate)
    else
        adaptive.SendRate = adaptive.ProbeRate
    end
    cycle.SampleAt = now
end

RebirthTab.performRebirth = function(generation)
    local adaptive = RebirthTab.adaptive
    local cycle = {
        Target = 5000 + RebirthTab.rebirths.Value * 2550,
        Rebirths = RebirthTab.rebirths.Value,
        SampleAt = os.clock(),
        StartStrength = RebirthTab.strength.Value,
        SampleStrength = RebirthTab.strength.Value,
        LastSentAt = os.clock(),
        Sent = 0,
        Tokens = 0,
        LastAt = os.clock(),
        StatusAt = -math.huge,
        TrainingSeconds = 0
    }
    adaptive.SendRate = adaptive.StrengthPerRep and math.clamp(math.max(0, cycle.Target - cycle.StartStrength) / adaptive.StrengthPerRep / adaptive.TargetSeconds, adaptive.MinRate, adaptive.MaxRate) or adaptive.ProbeRate
    cycle.RepBudget = adaptive.StrengthPerRep and math.max(adaptive.ProbeReps, math.ceil(math.max(0, cycle.Target - cycle.StartStrength) / adaptive.StrengthPerRep)) or adaptive.ProbeReps
    
    local lastTrainingAt = os.clock()
    local function needsTraining()
        return RebirthTab.isCurrent(generation) and RebirthTab.rebirths.Value == cycle.Rebirths and RebirthTab.strength.Value < cycle.Target
    end
    
    while needsTraining() do
        local now = os.clock()
        local character = lp.player.Character
        local humanoid = character and character:FindFirstChildOfClass("Humanoid")
        if humanoid and humanoid.Health > 0 then
            cycle.TrainingSeconds = cycle.TrainingSeconds + math.max(0, now - lastTrainingAt)
            RebirthTab.updateAdaptiveSpeed(cycle)
            if cycle.Sent >= cycle.RepBudget then
                cycle.Tokens = 0
                cycle.LastAt = now
                if now - cycle.LastSentAt >= adaptive.FeedbackTimeout then
                    local missing = math.max(0, cycle.Target - RebirthTab.strength.Value)
                    local extra = adaptive.StrengthPerRep and math.ceil(missing / adaptive.StrengthPerRep) or adaptive.ProbeReps
                    cycle.RepBudget = cycle.Sent + math.clamp(extra, math.ceil(adaptive.ProbeReps / 2), adaptive.ProbeReps)
                end
            else
                local budget = cycle.RepBudget - cycle.Sent
                local sent = sendTrainingReps(cycle, adaptive.SendRate, 5, function()
                    if budget <= 0 or not needsTraining() then return false end
                    budget = budget - 1
                    return true
                end)
                cycle.Sent = cycle.Sent + sent
                if sent > 0 then cycle.LastSentAt = now end
            end
            if now - cycle.StatusAt >= 0.25 then
                RebirthTab.Status.Text = (cycle.Sent >= cycle.RepBudget and "Status: Waiting for Strength Update | " or "Status: Strength Training | ") .. string.format("%.1fs", cycle.TrainingSeconds) .. " | " .. formatNumber(adaptive.SendRate) .. " reps/s"
                cycle.StatusAt = now
            end
        else
            cycle.Tokens = 0
            cycle.LastAt = now
            RebirthTab.Status.Text = "Status: Waiting for Respawn"
        end
        lastTrainingAt = now
        RunService.Heartbeat:Wait()
    end
    
    if not RebirthTab.isCurrent(generation) or RebirthTab.rebirths.Value ~= cycle.Rebirths then return end
    RebirthTab.updateAdaptiveSpeed(cycle, true)
    adaptive.LastTrainingSeconds = cycle.TrainingSeconds
    RebirthTab.Status.Text = "Status: Switching to Rebirth Pets"
    
    equipBestRebirthPets(function() return RebirthTab.isCurrent(generation) end)
    task.wait(0.25)
    
    while RebirthTab.isCurrent(generation) and RebirthTab.rebirths.Value == cycle.Rebirths do
        local remaining = RebirthTab.nextRequestAt - os.clock()
        if remaining > 0 then
            RebirthTab.Status.Text = "Status: Rebirth Cooldown | " .. string.format("%.1fs", remaining)
            task.wait(math.min(remaining, 0.1))
        else
            RebirthTab.nextRequestAt = os.clock() + RebirthTab.cooldown
            pcall(function()
                local remote = ReplicatedStorage:FindFirstChild("rEvents") and ReplicatedStorage.rEvents:FindFirstChild("rebirthRemote")
                if remote then
                    if remote:IsA("RemoteFunction") then
                        remote:InvokeServer("rebirthRequest")
                    elseif remote:IsA("RemoteEvent") then
                        remote:FireServer("rebirthRequest")
                    end
                end
            end)
            RebirthTab.recordRebirth()
        end
    end
end

RebirthTab.startWorker = function()
    if RebirthTab.workerActive then return end
    RebirthTab.workerActive = true
    runtime:Spawn(function(generation, success)
        success = pcall(function()
            while RebirthTab.isCurrent(generation) do
                equipBestFarmingPets(function() return RebirthTab.isCurrent(generation) end)
                if not RebirthTab.isCurrent(generation) then break end
                RebirthTab.performRebirth(generation)
                task.wait()
            end
        end)
        RebirthTab.workerActive = false
        if not success then task.wait(0.5) end
        if RebirthTab.running then
            RebirthTab.startWorker()
        end
    end, RebirthTab.generation)
end

RebirthTab.recordRebirth = function()
    local current = RebirthTab.rebirths.Value
    local previous = RebirthTab.statSamples.Rebirths
    if current > previous then
        RebirthTab.lastConfirmedAt = os.clock()
        RebirthTab.nextRequestAt = RebirthTab.lastConfirmedAt + RebirthTab.cooldown
        if RebirthTab.running then
            if not RebirthTab.baselineReady then
                RebirthTab.baselineReady = true
                RebirthTab.statSamples.Rebirths = current
                RebirthTab.timerStarted = true
                RebirthTab.startedAt = os.clock()
            else
                RebirthTab.statGains.Rebirths = RebirthTab.statGains.Rebirths + math.max(0, current - previous)
                RebirthTab.measuredCycles = RebirthTab.measuredCycles + 1
                if RebirthTab.measuredCycles >= 2 then
                    RebirthTab.updateAverage()
                end
            end
        end
    end
    RebirthTab.statSamples.Rebirths = current
    RebirthTab.updateStats()
end

local function formatExactNumber(number)
    local formatted = tostring(math.floor(math.abs(number)))
    local k
    while true do
        formatted, k = string.gsub(formatted, "^(-?%d+)(%d%d%d)", '%1,%2')
        if k == 0 then break end
    end
    return (number < 0 and "-" or "") .. formatted
end

local function formatRate(ratePerSecond)
    return formatNumber(ratePerSecond) .. "/s | " .. formatNumber(ratePerSecond * 60) .. "/min | " .. formatNumber(ratePerSecond * 3600) .. "/h | " .. formatNumber(ratePerSecond * 86400) .. "/d"
end

local startTime = tick()
local initialStrength = strengthStat.Value
local initialDurability = durabilityStat.Value
local initialRebirths = rebirthsStat.Value
local strengthHistory = {}
local durabilityHistory = {}
local rebirthsHistory = {}
local calculationInterval = 2

task.spawn(function()
    local lastCalcTime = tick()
    while true do
        local currentTime = tick()
        local currentStrength = strengthStat.Value
        local currentDurability = durabilityStat.Value
        local currentRebirths = rebirthsStat.Value
        
        local elapsedTime = currentTime - startTime
        local days = math.floor(elapsedTime / (24 * 3600))
        local hours = math.floor((elapsedTime % (24 * 3600)) / 3600)
        local minutes = math.floor((elapsedTime % 3600) / 60)
        local seconds = math.floor(elapsedTime % 60)
        stopwatchLabel.Text = string.format("Fast Rep Time: %dd %dh %dm %ds", days, hours, minutes, seconds)
        
        local sessionStrengthDelta = currentStrength - initialStrength
        local sessionDurabilityDelta = currentDurability - initialDurability
        local sessionRebirthsDelta = currentRebirths - initialRebirths
        
        strengthLabel.Text = "Strength: " .. formatNumber(currentStrength) .. " | Gained: " .. formatNumber(sessionStrengthDelta)
        durabilityLabel.Text = "Durability: " .. formatNumber(currentDurability) .. " | Gained: " .. formatNumber(sessionDurabilityDelta)
        rebirthsCountLabel.Text = "Rebirths: " .. formatExactNumber(currentRebirths) .. " | Gained: " .. formatExactNumber(sessionRebirthsDelta)
        
        table.insert(strengthHistory, {time = currentTime, value = currentStrength})
        table.insert(durabilityHistory, {time = currentTime, value = currentDurability})
        table.insert(rebirthsHistory, {time = currentTime, value = currentRebirths})
        
        while #strengthHistory > 0 and currentTime - strengthHistory[1].time > calculationInterval do table.remove(strengthHistory, 1) end
        while #durabilityHistory > 0 and currentTime - durabilityHistory[1].time > calculationInterval do table.remove(durabilityHistory, 1) end
        while #rebirthsHistory > 0 and currentTime - rebirthsHistory[1].time > calculationInterval do table.remove(rebirthsHistory, 1) end
        
        if currentTime - lastCalcTime >= calculationInterval then
            lastCalcTime = currentTime
            if #strengthHistory >= 2 then
                local strengthDelta = strengthHistory[#strengthHistory].value - strengthHistory[1].value
                local strengthPerSecond = strengthDelta / calculationInterval
                projectedStrengthLabel.Text = "Strength: " .. formatRate(strengthPerSecond)
            end
            if #durabilityHistory >= 2 then
                local durabilityDelta = durabilityHistory[#durabilityHistory].value - durabilityHistory[1].value
                local durabilityPerSecond = durabilityDelta / calculationInterval
                projectedDurabilityLabel.Text = "Durability: " .. formatRate(durabilityPerSecond)
            end
            if #rebirthsHistory >= 2 then
                local rebirthsDelta = rebirthsHistory[#rebirthsHistory].value - rebirthsHistory[1].value
                local rebirthsPerSecond = rebirthsDelta / calculationInterval
                rebirthsRateLabel.Text = "Rebirths Rate: " .. formatRate(rebirthsPerSecond)
            end
        end
        task.wait(0.2)
    end
end)

local function formatEventName(itemName)

    for word in itemName:gmatch("%S+") do table.insert(parts, word:lower()) end
    for i = 2, #parts do parts[i] = parts[i]:sub(1, 1):upper() .. parts[i]:sub(2) end
    return table.concat(parts)
end

local function activateRandomItems(count)

    for _, item in ipairs(itemList) do table.insert(shuffledItems, item) end
    for i = #shuffledItems, 2, -1 do
        local j = math.random(i)
        shuffledItems[i], shuffledItems[j] = shuffledItems[j], shuffledItems[i]
    end
    for i = 1, math.min(count, #shuffledItems) do
        local char = LP.Character
        local backpack = LP:FindFirstChild("Backpack")
        local tool = (char and char:FindFirstChild(shuffledItems[i])) or (backpack and backpack:FindFirstChild(shuffledItems[i]))
        if tool and LP:FindFirstChild("muscleEvent") then
            local eventName = formatEventName(shuffledItems[i])
            pcall(function() LP.muscleEvent:FireServer(eventName, tool) end)
        end
    end
end

    root.CFrame = CFrame.lookAt(targetPos, headPos)

            firetouchinterest(hitbox, rightHand, 0) 			
            firetouchinterest(hitbox, rightHand, 1) 			
            firetouchinterest(hitbox, leftHand, 0) 			
            firetouchinterest(hitbox, leftHand, 1) 		
        end 	end) 	
            
    if os.clock() - boss.Data.LastPunch >= boss.Data.PunchInterval then 		
        boss.Data.LastPunch = os.clock() 		
        pcall(function() 			
            if LP:FindFirstChild("muscleEvent") then

                LP.muscleEvent:FireServer("punch", "rightHand") 			
            end
            if punch and punch.Parent == character then 
                punch:Activate() 
            end 		end) 	end 	
            
    boss.BossLabel.Text = (boss.BossLabelPrefix or "Boss: ") .. boss.getBossDisplayName(model) 	
    boss.runtime.health = boss.getBossHealth(health, model) 	
    boss.runtime.maxHealth = boss.getBossMaxHealth(maxHealth, model) 	
    boss.HealthLabel.Text = "Health: " .. formatCurrentStat(boss.runtime.health) .. " / " .. formatCurrentStat(boss.runtime.maxHealth)

                root.AssemblyLinearVelocity = Vector3.zero 			end) 			
            task.wait(0.2)

        BossTab.Halloween.AutoFarmSwitch:Set(false) 	
    end 	
    BossTab.Data.Running = state 	
    BossTab.Data.Generation = BossTab.Data.Generation + 1 	
    BossTab.runtime.generation = BossTab.Data.Generation 	
    if not state then 		
        BossTab.Data.Phase = "Waiting" 		
        BossTab.StatusLabel.Text = "Status: Disabled" 		
        return 	end 	
    runtime:Spawn(function(generation, model, health, maxHealth, hitbox, selected) 		
        while BossTab.Data.Running and BossTab.Data.Generation == generation do 			
            pcall(function() 				
                model, health, maxHealth, hitbox = BossTab.findActiveBoss() 				
                selected = model and BossTab.shouldFarmBoss(model) 				
                if model and not selected then 					
                    BossTab.Data.Phase = "Waiting" 					
                    BossTab.StatusLabel.Text = "Status: Skipping " .. (workspace:GetAttribute("BossDisplayName") or model.Name) 				
                elseif model then 					
                    BossTab.Data.Phase = "Fighting" 					
                    if lp.player.Character and lp.player.Character:FindFirstChildOfClass("Humanoid") and lp.player.Character:FindFirstChildOfClass("Humanoid").Health > 0 then 						
                        BossTab.StatusLabel.Text = "Status: Fighting" 						
                        BossTab.attack(BossTab, model, health, maxHealth, hitbox) 					
                    else 						
                        BossTab.StatusLabel.Text = "Status: Waiting for Respawn" 					end 				
                else 					
                    BossTab.Data.Phase = "Waiting" 					
                    BossTab.StatusLabel.Text = "Status: Waiting for Boss" 				
                end 			
            end) 			
            task.wait(model and selected and 0.01 or 0.4) 		
        end 	end, BossTab.runtime.generation) 
end) 

-- ==================== HALLOWEEN BOSS FARMING ====================
BossTab:AddLabel("") 
BossTab:AddHeading("Halloween Boss Farming") 

BossTab.Halloween = { 	
    runtime = {}, 	
    BossLabelPrefix = "Halloween Boss: ", 	
    StatusLabel = BossTab:AddLabel("Status: Waiting for Halloween Boss"), 	
    BossLabel = BossTab:AddLabel("Halloween Boss: None"), 	
    HealthLabel = BossTab:AddLabel("Health: 0 / 0"), 	
    NextBossLabel = BossTab:AddLabel("Next Halloween Boss: Waiting for spawn"), 	
    Data = { 		
        Running = false, 		
        Generation = 0, 		
        Phase = "Waiting", 		
        PunchInterval = 0.01, 		
        LastPunch = 0, 		
        Bosses = { 			
            GrimReaper = "Grim Reaper", 			
            FrankenBrute = "Frankenbrute", 			
            PumpkinTitan = "Pumpkin Titan", 			
            ScarecrowColossus = "Scarecrow Colossus" 		}, 		
        BossOrder = {"GrimReaper", "FrankenBrute", "PumpkinTitan", "ScarecrowColossus"}, 		
        BossByDisplayName = {}, 		
        SelectedBosses = {}, 		
        Target = nil 	
    } 
} 

BossTab.Halloween.getBossDisplayName = function(model) 	
    return BossTab.Halloween.Data.Bosses[model:GetAttribute("BossId") or model.Name] or model.Name 
end 

BossTab.Halloween.getBossHealth = function(health, model) 	
    return tonumber(model:GetAttribute("Health")) or 0 
end 

BossTab.Halloween.getBossMaxHealth = function(maxHealth, model) 	
    return tonumber(model:GetAttribute("MaxHealth")) or 0 
end 

BossTab:AddSection("Selección Halloween Bosses")

for _, bossId in ipairs(BossTab.Halloween.Data.BossOrder) do 	
    local displayName = BossTab.Halloween.Data.Bosses[bossId]
    BossTab.Halloween.Data.BossByDisplayName[displayName] = bossId 	
    BossTab.Halloween.Data.SelectedBosses[bossId] = true 	
    
    BossTab:CreateToggle({
        Name = "Farm " .. displayName,
        Description = "Activar para farmear a " .. displayName,
        Default = true
    }, function(state)
        BossTab.Halloween.Data.SelectedBosses[bossId] = state
    end)
end 

BossTab.Halloween.shouldFarmBoss = function(model) 	
    return BossTab.Halloween.Data.SelectedBosses[model:GetAttribute("BossId") or model.Name] == true 
end 

BossTab.Halloween.findActiveBoss = function(selectedOnly, model, hitbox, candidateHitbox, character, root, distance, closestDistance) 	
    character = lp.player.Character 	
    root = character and character:FindFirstChild("HumanoidRootPart") 	
    closestDistance = math.huge 	
    for _, candidate in ipairs(services.collectionservice:GetTagged("HalloweenBoss")) do 		
        if candidate:IsA("Model") and candidate:IsDescendantOf(workspace) and BossTab.Halloween.getBossHealth(nil, candidate) > 0 and (not selectedOnly or BossTab.Halloween.shouldFarmBoss(candidate)) then 			
            candidateHitbox = candidate:FindFirstChild("BossDamageHitbox", true) or candidate.PrimaryPart or candidate:FindFirstChild("HumanoidRootPart") 			
            if candidateHitbox and candidateHitbox:IsA("BasePart") then 				
                if selectedOnly and candidate == BossTab.Halloween.Data.Target then 					
                    return candidate, nil, nil, candidateHitbox 				end 				
                distance = root and (root.Position - candidateHitbox.Position).Magnitude or 0 				
                if not model or distance < closestDistance then 					
                    model, hitbox, closestDistance = candidate, candidateHitbox, distance 				end 			
            end 		end 	end 	
    return model, nil, nil, hitbox 
end 

runtime:Spawn(function(model) 	
    while true do 		
        model = BossTab.Halloween.Data.Target 		
        if not model or not model.Parent or BossTab.Halloween.getBossHealth(nil, model) <= 0 then 			
            model = BossTab.Halloween.findActiveBoss() 		end 		
        if model then 			
            BossTab.Halloween.BossLabel.Text = "Halloween Boss: " .. BossTab.Halloween.getBossDisplayName(model) 			
            BossTab.Halloween.HealthLabel.Text = "Health: " .. formatCurrentStat(BossTab.Halloween.getBossHealth(nil, model)) .. " / " .. formatCurrentStat(BossTab.Halloween.getBossMaxHealth(nil, model)) 			
            BossTab.Halloween.NextBossLabel.Text = "Next Halloween Boss: Current boss active" 		
        else 			
            BossTab.Halloween.BossLabel.Text = "Halloween Boss: None" 			
            BossTab.Halloween.HealthLabel.Text = "Health: 0 / 0" 			
            BossTab.Halloween.NextBossLabel.Text = "Next Halloween Boss: Waiting for spawn" 		end 		
        task.wait(0.4) 	end 
end) 

BossTab.Halloween.AutoFarmSwitch = BossTab:AddSwitch("Auto Farm Halloween Boss", function(state) 	
    if state and BossTab.Data.Running then BossTab.AutoFarmSwitch:Set(false) end 	
    BossTab.Halloween.Data.Running = state 	
    BossTab.Halloween.Data.Generation = BossTab.Halloween.Data.Generation + 1 	
    if not state then 		
        BossTab.Halloween.Data.Target = nil 		
        BossTab.Halloween.Data.Phase = "Waiting" 		
        BossTab.Halloween.StatusLabel.Text = "Status: Disabled"

    runtime:Spawn(function(generation, model, health, maxHealth, hitbox)

            pcall(function()

                    BossTab.Halloween.Data.Phase = "Fighting"

                        BossTab.Halloween.StatusLabel.Text = "Status: Fighting" 						
                        BossTab.attack(BossTab.Halloween, model, health, maxHealth, hitbox) 					
                    else 						
                        BossTab.Halloween.StatusLabel.Text = "Status: Waiting for Respawn" 					end 				
                else 					
                    BossTab.Halloween.Data.Target = nil 					
                    BossTab.Halloween.Data.Phase = "Waiting" 					
                    BossTab.Halloween.StatusLabel.Text = "Status: Waiting for Halloween Boss" 				end
